    <p>Every mechanic here is drawn from <em>The World Climb</em>. Nothing was invented for gamification — it was already true.</p>
    <div class="pull">"Barter did not die when money arrived. Money emerged because trade expanded beyond what memory and reputation alone could efficiently track. Money is a tool for scaling the deal. It is not the beginning of the deal." — Chapter 1</div>
    <p><b style="color:var(--gold);">Information asymmetry</b> is the whole game. In real trade, you never see the other side's balance sheet, their creditors, their needs. You have to read the room. Every offer is a signal — patient offers versus urgent ones — and the skill is telling them apart.</p>
    <p><b style="color:var(--gold);">The Dignity of the Deal</b> is not a rule imposed on trade. It is the ethic that makes trade repeatable. A community where every deal exploits the desperate becomes a community where no one trusts a deal. Chapter 13 argues that the person who honors dignity in every trade builds a reputation ledger that pays out for years.</p>
    <p><b style="color:var(--gold);">The Coincidence of Wants</b> is barter's structural limit. It is why the Frankenstein Exchange exists as a mutual credit ledger, not a barter board. You do not have to find a perfect match; the ledger tracks your net contribution.</p>

  { text:"seems focused", trueDesperationRange:[10, 60] },      // mid signal, mixed
  { text:"seems in a hurry", trueDesperationRange:[40, 100] }   // high signal, high desperation
];

// ==================== STATE ====================
let state = {
  cur: 0,
  trades: 0,
  dignity: 100,
  reads: 3,
  offers: [],
  haggleUsed: false
};

// ==================== HELPERS ====================
function $(id) { return document.getElementById(id); }

function generateOffer() {
  // Each offer: a tier value (what they'd give you) and a signal + hidden trueDesperation
  const up = Math.floor(Math.random() * 3) + 1;  // +1 to +3
  let tier = Math.min(TOP, state.cur + up);
  // Small chance of lateral or slight downgrade
  const roll = Math.random();
  if (roll < 0.15) tier = Math.max(0, state.cur - 1 + Math.floor(Math.random() * 2));
  const signalIdx = Math.floor(Math.random() * SIGNALS.length);
  const signal = SIGNALS[signalIdx];
  const trueDesp = signal.trueDesperationRange[0] + Math.floor(Math.random() * (signal.trueDesperationRange[1] - signal.trueDesperationRange[0] + 1));
  return { tier, signalIdx, trueDesperation: trueDesp, revealed: false };
}

function generateOffers() {
  state.offers = [generateOffer(), generateOffer(), generateOffer()];
  state.haggleUsed = false;
}

function renderLadder() {
  const el = $('ladder');
  let html = '';
  for (let i = 0; i < LADDER.length; i++) {
    html += `<div class="rung ${i <= state.cur ? 'on' : ''}" title="${LADDER[i].nm}"></div>`;
  }
  el.innerHTML = html;
}

function renderHolding() {
  $('holdingEm').textContent = LADDER[state.cur].em;
  $('holdingNm').textContent = LADDER[state.cur].nm;
}

function renderHUD() {
  $('hudRung').textContent = (state.cur + 1) + ' / ' + LADDER.length;
  $('hudTrades').textContent = state.trades;
  $('hudDignity').textContent = state.dignity;
  $('hudReads').textContent = state.reads;
  const dCell = $('hudDignityCell');
  dCell.classList.remove('warn', 'bad');
  if (state.dignity < 50) dCell.classList.add('bad');
  else if (state.dignity < 75) dCell.classList.add('warn');
}

function renderOffers() {
  const el = $('offers');
  let html = '';
  state.offers.forEach((o, i) => {
    const item = LADDER[o.tier];
    const signal = SIGNALS[o.signalIdx];
    const revealClass = o.revealed ? (o.trueDesperation >= 50 ? 'revealed desperate' : 'revealed') : '';
    const revealedText = o.trueDesperation >= 70 ? 'Desperate — needs your hour badly'
                       : o.trueDesperation >= 50 ? 'Under pressure — will take a rushed deal'
                       : o.trueDesperation >= 25 ? 'Willing but reasonable'
                       : 'Truly patient — no urgency';
    html += `
      <div class="offer ${revealClass}" data-i="${i}">
        <div class="offer-em">${item.em}</div>
        <div class="offer-nm">${item.nm}</div>
        <div class="offer-signal">${signal.text}</div>
        <div class="offer-revealed">${revealedText}</div>
      </div>`;
  });
  el.innerHTML = html;
  el.querySelectorAll('.offer').forEach(card => {
    card.addEventListener('click', () => {
      const i = parseInt(card.dataset.i, 10);
      if (readMode) revealOffer(i);
      else takeOffer(i);
    });
  });
  $('btnHaggle').disabled = state.haggleUsed;
  $('btnReadRoom').disabled = state.reads === 0;
}

// ==================== ACTIONS ====================
let readMode = false;

function takeOffer(i) {
  const o = state.offers[i];
  state.trades++;
  const oldTier = state.cur;
  const newTier = o.tier;
  const direction = newTier > oldTier ? 'up' : newTier < oldTier ? 'down' : 'flat';

  // Dignity impact based on true desperation and whether player took an obvious upgrade from a desperate person
  let dignityChange = 0;
  let toastMsg = '';
  const jumpSize = newTier - oldTier;

  if (o.trueDesperation >= 70 && jumpSize >= 2) {
    // Exploited someone desperate for significant jump
    dignityChange = -15;
    toastMsg = `You climbed to ${LADDER[newTier].nm} — but they were desperate. The deal was uneven.`;
  } else if (o.trueDesperation >= 50 && jumpSize >= 3) {
    // Modest exploitation
    dignityChange = -8;
    toastMsg = `You climbed to ${LADDER[newTier].nm}. They were under pressure. It sits with you.`;
  } else if (o.trueDesperation < 25 && jumpSize >= 2) {
    // Fair trade with patient counterparty (best case)
    dignityChange = +5;
    toastMsg = `A fair swap for ${LADDER[newTier].nm}. Both parties walked away right.`;
  } else if (direction === 'down') {
    // You traded down; sometimes the right move
    dignityChange = +2;
    toastMsg = `You traded down to ${LADDER[newTier].nm}. Sometimes you have to give ground to gain later.`;
  } else if (direction === 'flat') {
    dignityChange = 0;
    toastMsg = `Lateral swap. The market gave you a sideways deal.`;
  } else {
    dignityChange = 0;
    toastMsg = `Traded up to ${LADDER[newTier].nm}. A working deal.`;
  }

  state.dignity = Math.max(0, Math.min(100, state.dignity + dignityChange));
  state.cur = newTier;

  $('toast').textContent = toastMsg;

  if (state.cur >= TOP) return endGame();

  generateOffers();
  renderHolding(); renderLadder(); renderHUD(); renderOffers();
}

function revealOffer(i) {
  if (state.offers[i].revealed) return;
  state.reads--;
  state.offers[i].revealed = true;
  readMode = false;
  $('btnReadRoom').classList.remove('ghost');
  $('btnReadRoom').classList.add('ghost');
  $('toast').textContent = 'Read the room — one signal was true, or false. Now decide.';
  renderHUD(); renderOffers();
}

function haggle() {
  if (state.haggleUsed) return;
  state.haggleUsed = true;
  // Small dignity nick — you made them wait
  state.dignity = Math.max(0, state.dignity - 2);
  generateOffers();
  // haggleUsed stays true until next turn — one haggle per rung
  state.haggleUsed = true;
  renderHUD(); renderOffers();
  $('toast').textContent = 'You haggled. Three fresh counterparties are now at the table.';
  $('btnHaggle').disabled = true;
}

$('btnReadRoom').addEventListener('click', () => {
  if (state.reads === 0) return;
  readMode = !readMode;
  $('toast').textContent = readMode ? 'Read the room mode — click a counterparty to reveal their true desperation.' : '';
});

$('btnHaggle').addEventListener('click', haggle);
$('btnAgain').addEventListener('click', resetGame);

// ==================== ENDGAME + CERTIFICATE ====================
async function endGame() {
  renderHolding(); renderLadder(); renderHUD();

  // Determine tier
  let tier = 'Bronze';
  if (state.trades <= 7 && state.dignity >= 90) tier = 'Gold';
  else if (state.trades <= 8 && state.dignity >= 70) tier = 'Silver';
  else if (state.trades <= 10 && state.dignity >= 50) tier = 'Bronze';
  else tier = 'Participation';

  const score = Math.max(0, (1000 - (state.trades - 3) * 60) + state.dignity * 3);
  const now = new Date();
  const serial = 'SB-' + now.getFullYear() + String(now.getMonth() + 1).padStart(2, '0') + String(now.getDate()).padStart(2, '0') + '-' + Math.random().toString(36).slice(2, 6).toUpperCase();

  $('gameSection').style.display = 'none';
  $('endgame').classList.add('show');
  $('endCert').textContent = tier === 'Participation'
    ? 'The Deal Was Made — Try Again for a Certificate'
    : tier + ' Certificate of Sovereign Barter';
  $('endSerial').textContent = tier === 'Participation' ? '' : 'Serial: ' + serial;
  $('endTrades').textContent = state.trades;
  $('endDignity').textContent = state.dignity;
  $('endScore').textContent = score;

  // Save cert to Firebase
  if (db && uid && tier !== 'Participation') {
    try {
      await setDoc(doc(db, 'sovereign_barter_certs', serial), {
        tier, serial, trades: state.trades, dignity: state.dignity, score, uid, issuedAt: Date.now()
      }, { merge: true });
    } catch (e) {}
  }
}

function resetGame() {
  state = { cur:0, trades:0, dignity:100, reads:3, offers:[], haggleUsed:false };
  readMode = false;
  $('endgame').classList.remove('show');
  $('gameSection').style.display = '';
  $('toast').textContent = '';
  generateOffers();
  renderHolding(); renderLadder(); renderHUD(); renderOffers();
}

// ==================== RATING WIDGET ====================
let db = null, auth = null, uid = null;
const starsEl = document.getElementById('stars'), metaEl = document.getElementById('rateMeta'), thxEl = document.getElementById('rateThanks');
let myStars = 0, avg = 0, count = 0;

function paint(hover) {
  starsEl.innerHTML = '';
  const fill = hover || myStars || Math.round(avg);
  for (let i = 1; i <= 5; i++) {
    const s = document.createElement('span');
    s.className = 'st' + ((i <= fill) ? ' on' : '');
    s.textContent = '★';
    s.setAttribute('role', 'radio');
    s.addEventListener('mouseenter', () => paint(i));
    s.addEventListener('mouseleave', () => paint(0));
    s.addEventListener('click', () => rate(i));
    starsEl.appendChild(s);
  }
}
function showMeta() {
  if (count > 0) metaEl.innerHTML = '<b>' + avg.toFixed(1) + '</b> / 5 · ' + count + (count === 1 ? ' rating' : ' ratings');
  else metaEl.textContent = 'Be the first to rate this page.';
}
function injectLD() {
  if (count < 1) return;
  const el = document.getElementById('course-ld');
  if (!el) return;
  try {
    const o = JSON.parse(el.textContent);
    o.aggregateRating = { "@type": "AggregateRating", "ratingValue": Number(avg.toFixed(2)), "ratingCount": count, "bestRating": 5, "worstRating": 1 };
    el.textContent = JSON.stringify(o);
  } catch (e) {}
}
async function loadRating() {
  try {
    const snap = await getDoc(doc(db, 'lesson_ratings', GID));
    if (snap.exists()) { const d = snap.data(); count = d.count || 0; avg = count ? (d.sum || 0) / count : 0; }
  } catch (e) {}
  try {
    const v = await getDoc(doc(db, 'lesson_ratings', GID, 'voters', uid));
    if (v.exists()) myStars = v.data().stars || 0;
  } catch (e) {}
  paint(0); showMeta(); injectLD();
  if (myStars) thxEl.textContent = 'You rated this ' + myStars + ' ★. Tap to change.';
}
async function rate(stars) {
  if (!db || !uid) { thxEl.textContent = 'Ratings save when online.'; myStars = stars; paint(0); return; }
  try {
    await runTransaction(db, async (tx) => {
      const aggRef = doc(db, 'lesson_ratings', GID), vRef = doc(db, 'lesson_ratings', GID, 'voters', uid);
      const a = await tx.get(aggRef), v = await tx.get(vRef);
      let sum = a.exists() ? (a.data().sum || 0) : 0, c = a.exists() ? (a.data().count || 0) : 0;
      if (v.exists()) sum = sum - (v.data().stars || 0) + stars;
      else { sum += stars; c += 1; }
      tx.set(aggRef, { sum, count: c }, { merge: true });
      tx.set(vRef, { stars, at: Date.now() }, { merge: true });
    });
    myStars = stars;
    const snap = await getDoc(doc(db, 'lesson_ratings', GID));
    if (snap.exists()) { const d = snap.data(); count = d.count || 0; avg = count ? (d.sum || 0) / count : 0; }
    paint(0); showMeta(); injectLD();
    thxEl.textContent = 'Thank you — your rating was saved.';
  } catch (e) { thxEl.textContent = 'Could not save just now. Please try again.'; }
}
